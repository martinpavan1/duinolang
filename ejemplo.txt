const int sensorTemp = A0;
const int sensorX = A1;
const int led = 13;

void setup() {
  pinMode(led, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  // Leer el valor analógico (0-1023)
  int lectura = analogRead(sensorTemp);

  // Convertir a voltaje
  float voltaje = lectura * (5.0 / 1023.0);

  // LM35: 10 mV por cada °C
  float temperatura = voltaje * 100.0;

  // Mostrar la temperatura
  Serial.print("Temperatura: ");
  Serial.print(temperatura);
  Serial.println(" °C");

  // Si supera los 30 °C, encender el LED
  if (temperatura > 30.0) {
    digitalWrite(led, HIGH);
  } else {
    digitalWrite(led, LOW);
  }

  delay(1000);
}
